<script>
  (function () {
    var path = location.pathname.replace(/\/$/, "");
    var box = document.querySelector('[data-rh-route="' + path + '"]');
    if (!box) return;
    box.hidden = false;
    document.querySelector('[data-rh-route="404"]').remove();
    // остальные скрытые страницы убираем, чтобы скрипты и формы не видели лишнего
    Array.prototype.forEach.call(document.querySelectorAll("[data-rh-route][hidden]"), function (el) { el.remove(); });
    var h1 = box.querySelector("h1"); if (h1) document.title = h1.textContent.trim() + " — Retoria Home";
  })();
</script>
